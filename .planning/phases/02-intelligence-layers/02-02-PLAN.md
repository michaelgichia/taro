<task type="auto">
  <name>Create element analyzer for accessibility properties</name>
  <files>src/analyzer/visual/element-analyzer.ts</files>
  <action>
    Create src/analyzer/visual/element-analyzer.ts that:
    - Exports async function analyzeElementProperties(page: Page, selector: string): Promise<AccessibilityProperties>
    - Extracts: role, label, name, placeholder, testId, value
    - Identifies best query strategy: getByRole, getByLabelText, getByText, getByTestId
    - Ranks query strategies by robustness
    
    interface AccessibilityProperties {
      preferredQuery: QueryStrategy;
      alternatives: QueryStrategy[];
      hasAccessibleName: boolean;
      isInteractive: boolean;
    }
    
    interface QueryStrategy {
      method: string;
      args: (string | RegExp)[];
      priority: number; // 1 = best
    }
    
    Priority order:
    1. getByRole + name (most robust)
    2. getByLabelText (form fields)
    3. getByPlaceholderText
    4. getByAltText (images)
    5. getByTestId (if data-testid present)
    6. getByText (last resort)
  </action>
  <verify>npm test -- --grep "element-analyzer" passes</verify>
  <done>analyzeElementProperties returns ranked query strategies</done>
</task>
